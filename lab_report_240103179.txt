### 4. Forensic Explanation

The experiment measured performance using different memory strides.

The Apple M3 has a cache line size of 128 bytes. Therefore, stride 16 corresponds to 128 bytes of separation between worker slots.

However, the measurements did not reveal a clear performance improvement at stride 16.

At 8 processes, stride 1 achieved 535.0578 ms, while stride 16 achieved 550.3224 ms.

For stride 1, e(p) increases from 0.065551 at p=2 to 0.102490 at p=4 and 0.135828 at p=8. For stride 16, it increases from 0.081895 to 0.101478 and 0.144273. Both series show increasing effective overhead; the isolated layout does not produce a lower p=8 value. Karp-Flatt e(p) combines serial work, launch/synchronization costs and other scaling losses; it does not independently measure cache-coherence traffic.

This suggests that the experiment did not demonstrate a measurable reduction in false-sharing overhead. Python interpreter overhead, process scheduling and measurement variability may have obscured the cache-coherence effect.

No sharp byte-offset inflection was resolved in this sweep, so 128 bytes is the OS-reported cache-line width rather than an empirically detected boundary. At p=8, stride 16 is 2.85% slower than stride 1, with IQRs of 19.6698 ms and 14.7299 ms respectively. These data do not establish a false-sharing penalty. The measured result is retained rather than forcing the expected hardware effect.

## Source appendix: harness.py

````python


from dataclasses import dataclass
import multiprocessing as mp
import time
from typing import Callable


@dataclass
class BenchStats:
    min_ms: float
    median_ms: float
    iqr_ms: float
    max_ms: float


def fnv1a_hash(s: str) -> int:
    h = 14695981039346656037

    for b in s.encode("utf-8"):
        h ^= b
        h = (h * 1099511628211) & 0xFFFFFFFFFFFFFFFF

    return h


def run_benchmark(
    fn: Callable[[], None],
    warmup_runs: int = 3,
    timed_runs: int = 9
) -> BenchStats:

    for _ in range(warmup_runs):
        fn()

    samples = []

    for _ in range(timed_runs):
        t0 = time.perf_counter_ns()
        fn()
        t1 = time.perf_counter_ns()

        samples.append((t1 - t0) / 1e6)

    samples.sort()

    mid = timed_runs // 2

    if timed_runs % 2 == 1:
        med = samples[mid]
    else:
        med = 0.5 * (samples[mid - 1] + samples[mid])

    q25 = samples[timed_runs // 4]
    q75 = samples[(3 * timed_runs) // 4]

    return BenchStats(
        samples[0],
        med,
        q75 - q25,
        samples[-1]
    )


def emit_verify_tag(
    task_id: str,
    student_seed: int,
    primary_metric: float,
    secondary_metric: float
) -> str:

    hw = mp.cpu_count()

    payload = (
        f"{task_id}|{student_seed}|{hw}|"
        f"{primary_metric:.4f}|{secondary_metric:.4f}"
    )

    sig = fnv1a_hash(payload)

    return (
        f"VERIFY_HASH [{task_id}]: "
        f"{sig:016X} (P_log={hw})"
    )



````

N_SLOTS = 512
SHM_SIZE = 4096

STRIDES = (1, 2, 4, 8, 16, 32)


# ==========================================
# WORKER FUNCTION
# ==========================================

def fs_worker(shm_name, slot_idx, iters):

    shm = shared_memory.SharedMemory(name=shm_name)

    arr = None

    try:
        arr = (ctypes.c_uint64 * N_SLOTS).from_buffer(shm.buf)

        acc = arr[slot_idx]

        mask = 0xFFFFFFFFFFFFFFFF
        mult = 6364136223846793005

        for i in range(iters):

            acc = (acc * mult + (i | 1)) & mask

            arr[slot_idx] = acc

    finally:

        # Release exported buffer before closing shared memory
        if arr is not None:
            del arr

        shm.close()


# ==========================================
# SINGLE BENCHMARK TRIAL
# ==========================================

def run_trial(p, stride):

    if TOTAL_OPS % p != 0:
        raise ValueError("TOTAL_OPS must be divisible by p")

    if (p - 1) * stride >= N_SLOTS:
        raise ValueError("Shared memory slot index is out of range")

    shm = shared_memory.SharedMemory(
        create=True,
        size=SHM_SIZE
    )

    processes = []

    try:

        # Initialize the worker slots safely.
        # Avoid memoryview slice assignment.
        slots = (ctypes.c_uint64 * N_SLOTS).from_buffer(shm.buf)

        try:
            for t in range(p):
                slots[t * stride] = 0
        finally:
            del slots

        iters = TOTAL_OPS // p

        ctx = mp.get_context("spawn")

        # Start workers
        for t in range(p):

            slot_idx = t * stride

            proc = ctx.Process(
                target=fs_worker,
                args=(shm.name, slot_idx, iters)
            )

            proc.start()
            processes.append(proc)

        # Wait for workers
        for proc in processes:
            proc.join()

        # Check worker exit codes
        failed = [
            (proc.pid, proc.exitcode)
            for proc in processes
            if proc.exitcode != 0
        ]

        if failed:
            raise RuntimeError(
                f"Worker process failed: {failed}"
            )

    finally:

        # Stop any remaining processes
        for proc in processes:

            if proc.is_alive():
                proc.terminate()
                proc.join()

        # Release shared memory
        shm.close()
        shm.unlink()


# ==========================================
# BENCHMARK MEASUREMENT
# ==========================================

def measure(p, stride):

    print(
        f"Measuring: processes={p}, stride={stride}",
        flush=True
    )

    stats = run_benchmark(

        lambda: run_trial(p, stride),

    )

"""Phase 2: Strong/weak scaling benchmark for the parallel computing lab.

Uses only Python's standard library and the existing harness.py module.
Run from the directory containing harness.py:
    python3 task2_scaling.py
"""

import math
import multiprocessing as mp
import statistics
import time

from harness import emit_verify_tag, run_benchmark


# ---------------------------------------------------------------------------
# Configuration from the lab handout and the student's Phase 0 measurements
# ---------------------------------------------------------------------------
STUDENT_ID = 240103149
P_LOG = mp.cpu_count()

N_STRONG = 1_600_000
N0_WEAK = 250_000
SERIAL_PREFIX_FRACTION = 0.05
SERIAL_TRANSCENDENTAL_PAIRS = 6  # 6 * (sin + sqrt) = 12 operations / element
PARALLEL_ITERATIONS = 35

WARMUP_RUNS = 2
TIMED_RUNS = 7


# ---------------------------------------------------------------------------
# Stage A: serial work in the parent process
# ---------------------------------------------------------------------------
def serial_stage(n):
    """Process the first 5% of n elements using 12 transcendental ops each."""
    prefix_n = max(1, int(n * SERIAL_PREFIX_FRACTION))
    total = 0.0

    for i in range(prefix_n):
        x = (i + 1) * 1.0e-6 + 0.25
        for _ in range(SERIAL_TRANSCENDENTAL_PAIRS):
            x = math.sin(x)
            x = math.sqrt(abs(x) + 1.0e-12)
        total += x

    return total


# ---------------------------------------------------------------------------
# Stage B: parallel worker; must be top-level for macOS 'spawn'
# ---------------------------------------------------------------------------
def parallel_worker(bounds):
    """Compute one contiguous chunk with 35 multiply-add/trig iterations."""
    start, end = bounds
    total = 0.0

    for i in range(start, end):
        x = (i % 1000 + 1) * 1.0e-4
        for k in range(PARALLEL_ITERATIONS):
            # One multiply-add style update and one transcendental operation.
            x = x * 1.0000001 + math.sin(x + k * 1.0e-6) * 1.0e-4
        total += x

    return total


def make_even_chunks(n, p):
    """Return p contiguous, evenly sized (or near-even) non-overlapping chunks."""
    base, remainder = divmod(n, p)
    chunks = []
    start = 0

    for worker_id in range(p):
        chunk_size = base + (1 if worker_id < remainder else 0)
        end = start + chunk_size
        chunks.append((start, end))
        start = end

    return chunks


# ---------------------------------------------------------------------------
# One complete pipeline run; the Pool itself is created outside this function
# and remains alive for all warm-up and timed repetitions at a given p.
# ---------------------------------------------------------------------------
def run_pipeline(pool, n, p, stage_a_samples):
    stage_start = time.perf_counter_ns()
    serial_result = serial_stage(n)
    stage_end = time.perf_counter_ns()

    stage_a_samples.append((stage_end - stage_start) / 1.0e6)

    chunks = make_even_chunks(n, p)
    partial_results = pool.map(parallel_worker, chunks, chunksize=1)

    # Consume the deterministic results so the complete computation is used.
    checksum = serial_result + sum(partial_results)
    if not math.isfinite(checksum):
        raise ArithmeticError("The pipeline produced a non-finite result")


def measure_pipeline(ctx, p, n, label):
    """Create a persistent pool, then benchmark the pipeline with it."""
    print(f"Measuring {label}: p={p}, N={n:,}", flush=True)
    stage_a_samples = []

    # Pool creation/startup is outside run_benchmark's timed region.
    with ctx.Pool(processes=p) as pool:
        stats = run_benchmark(
            lambda: run_pipeline(pool, n, p, stage_a_samples),
            warmup_runs=WARMUP_RUNS,
            timed_runs=TIMED_RUNS,
        )

    # run_benchmark makes 2 warm-ups followed by 7 timed invocations.
    timed_stage_samples = stage_a_samples[-TIMED_RUNS:]
    stage_a_median_ms = statistics.median(timed_stage_samples)

    return stats, stage_a_median_ms


# ---------------------------------------------------------------------------
# Reporting helpers
# ---------------------------------------------------------------------------
def print_strong_table(results):
    t1 = results[1][0].median_ms

    print("\nTABLE 2A: STRONG SCALING")
    print("-" * 82)
    print(
        f"{'p':<5}{'N':<14}{'Median (ms)':<16}"
        f"{'IQR (ms)':<13}{'Speedup':<13}{'Efficiency':<13}"
    )

    for p in sorted(results):
        stats, _stage_a = results[p]
        speedup = t1 / stats.median_ms
        efficiency = speedup / p
        print(
            f"{p:<5}{N_STRONG:<14,}{stats.median_ms:<16.4f}"
            f"{stats.iqr_ms:<13.4f}{speedup:<13.4f}{efficiency:<13.4f}"
        )


def print_weak_table(results):
    t1_weak = results[1][0].median_ms

    print("\nTABLE 2B: WEAK SCALING")
    print("-" * 90)
    print(
        f"{'p':<5}{'N(p)':<14}{'Median (ms)':<16}{'IQR (ms)':<13}"
        f"{'Efficiency':<14}{'Alpha':<12}{'Gustafson S':<14}"
    )

    for p in sorted(results):
        stats, stage_a_ms = results[p]
        n = p * N0_WEAK
        efficiency = t1_weak / stats.median_ms
        alpha = stage_a_ms / stats.median_ms
        gustafson = p - alpha * (p - 1)

        print(
            f"{p:<5}{n:<14,}{stats.median_ms:<16.4f}"
            f"{stats.iqr_ms:<13.4f}{efficiency:<14.4f}"
            f"{alpha:<12.6f}{gustafson:<14.4f}"
        )


def print_marginal_speedup(results):
    t1 = results[1][0].median_ms
    previous_gain = None

    print("\nMARGINAL SPEEDUP PER ADDED WORKER")
    print("-" * 66)
    print(f"{'Transition':<18}{'Added speedup':<20}{'Change vs prior gain (%)':<26}")

    for p in range(2, P_LOG + 1):
        previous_speedup = t1 / results[p - 1][0].median_ms
        current_speedup = t1 / results[p][0].median_ms
        current_gain = current_speedup - previous_speedup

        if previous_gain is None or abs(previous_gain) < 1.0e-12:
            change_text = "N/A"
        else:
            change = (current_gain - previous_gain) / abs(previous_gain) * 100.0
            change_text = f"{change:.2f}"

        print(f"{p - 1} -> {p:<13}{current_gain:<20.4f}{change_text:<26}")
        previous_gain = current_gain


# ---------------------------------------------------------------------------
# Main experiment
# ---------------------------------------------------------------------------
def main():
    if P_LOG < 1:
        raise RuntimeError(f"Invalid logical CPU count: {P_LOG}")

    ctx = mp.get_context("spawn")

    print("=" * 70)
    print("PHASE 2: STRONG SCALING AND WEAK SCALING")
    print("=" * 70)
    print(f"Student ID: {STUDENT_ID}")
    print(f"Logical CPUs: {P_LOG}")
    print(f"Strong scaling N: {N_STRONG:,}")
    print(f"Weak scaling N0 per process: {N0_WEAK:,}")
    print(f"Serial prefix: {SERIAL_PREFIX_FRACTION:.0%}")
    print(f"Parallel iterations per element: {PARALLEL_ITERATIONS}")
    print(f"Benchmark repetitions: {WARMUP_RUNS} warm-up + {TIMED_RUNS} timed")

    # Experiment 2A: fixed N, every integer process count from 1 to P_LOG.
    strong_results = {}
    print("\nStarting Experiment 2A: Strong Scaling", flush=True)
    for p in range(1, P_LOG + 1):
        strong_results[p] = measure_pipeline(
            ctx, p, N_STRONG, "strong scaling"
        )

    print_strong_table(strong_results)

    # Experiment 2B: fixed work per worker; total N grows with p.
    weak_results = {}
    print("\nStarting Experiment 2B: Weak Scaling", flush=True)
    for p in range(1, P_LOG + 1):
        n = p * N0_WEAK
        weak_results[p] = measure_pipeline(
            ctx, p, n, "weak scaling"
        )

    print_weak_table(weak_results)

    # Amdahl analysis from the p=1 strong-scaling run.
    t1_strong = strong_results[1][0].median_ms
    t_stage_a_1 = strong_results[1][1]
    serial_fraction = t_stage_a_1 / t1_strong

    if not 0.0 < serial_fraction <= 1.0:
        raise ArithmeticError(
            f"Unexpected measured serial fraction: {serial_fraction}"
        )

    amdahl_max = 1.0 / serial_fraction
    amdahl_at_plog = 1.0 / (
        serial_fraction + (1.0 - serial_fraction) / P_LOG
    )
    observed_at_plog = t1_strong / strong_results[P_LOG][0].median_ms

    print("\nAMDAHL'S LAW ANALYSIS")
    print("-" * 55)
    print(f"Stage A median at p=1:       {t_stage_a_1:.4f} ms")
    print(f"Total median at p=1:         {t1_strong:.4f} ms")
    print(f"Measured serial fraction s:  {serial_fraction:.6f}")
    print(f"Theoretical maximum S(inf):  {amdahl_max:.4f}")
    print(f"Theoretical speedup at p={P_LOG}: {amdahl_at_plog:.4f}")
    print(f"Observed speedup at p={P_LOG}:   {observed_at_plog:.4f}")

    print_marginal_speedup(strong_results)

    # Gustafson scaled speedup at the largest worker count.
    weak_last_stats, weak_last_stage_a = weak_results[P_LOG]
    weak_last_alpha = weak_last_stage_a / weak_last_stats.median_ms
    gustafson_at_plog = P_LOG - weak_last_alpha * (P_LOG - 1)

    print("\nGUSTAFSON ANALYSIS")
    print("-" * 55)
    print(f"Stage A median at p={P_LOG}: {weak_last_stage_a:.4f} ms")
    print(f"Total weak median at p={P_LOG}: {weak_last_stats.median_ms:.4f} ms")
    print(f"Alpha at p={P_LOG}: {weak_last_alpha:.6f}")
    print(f"Gustafson scaled speedup: {gustafson_at_plog:.4f}")

    print("\nPHASE 2 VERIFICATION")
    print("-" * 55)
    print(
        emit_verify_tag(
            "PHASE2",
            STUDENT_ID,
            observed_at_plog,
            gustafson_at_plog,
        )
    )
    print("\nPHASE 2 COMPLETED")


if __name__ == "__main__":
    mp.freeze_support()
    main()

HERE = Path(__file__).resolve().parent

def log(message):

def save():

def verify(phase, a, b, logical):
    payload = f'{phase}|{SID}|{logical}|{a:.4f}|{b:.4f}'
    h = 14695981039346656037
    for byte in payload.encode('utf-8'):

def bench(fn, warmups, runs):
    for _ in range(warmups):
        fn()
    samples = []
    for _ in range(runs):
        start = time.perf_counter_ns()
        fn()
        samples.append((time.perf_counter_ns() - start) / 1e6)
    xs = sorted(samples)
    return dict(median_ms=statistics.median(xs),
                iqr_ms=xs[3*len(xs)//4] - xs[len(xs)//4], samples_ms=samples)


def sysctl(name):
    try:
        return subprocess.check_output(['sysctl', '-n', name], text=True,
                                       stderr=subprocess.DEVNULL).strip()
    except (OSError, subprocess.CalledProcessError):
        return 'unavailable'


def hardware():
    hw = {'OS': platform.platform(), 'Python': sys.version.split()[0],
          'logical_cpus': mp.cpu_count()}
    for key in ('machdep.cpu.brand_string', 'hw.physicalcpu', 'hw.cachelinesize',
                'hw.l1dcachesize', 'hw.l2cachesize', 'hw.l3cachesize',
                'hw.perflevel0.physicalcpu', 'hw.perflevel1.physicalcpu',
                'hw.perflevel0.l1dcachesize', 'hw.perflevel1.l1dcachesize',
                'hw.perflevel0.l2cachesize', 'hw.perflevel1.l2cachesize'):
        hw[key] = sysctl(key)
    return hw


_HANDLES = []
_VIEWS = []
_N = 0


def close_views():
    for view in _VIEWS:
        view.release()
    _VIEWS.clear()
    for handle in _HANDLES:
        handle.close()
    _HANDLES.clear()


def attach(names, fmt, n=0):
    global _N
    _N = n
    for name in names:
        handle = shared_memory.SharedMemory(name=name)
        _HANDLES.append(handle)
        _VIEWS.append(handle.buf.cast(fmt))
    atexit.register(close_views)


def ready(_):
    return True


def copy_job(bounds):
    lo, hi = bounds
    _VIEWS[0][lo:hi] = _VIEWS[1][lo:hi]


def compute_job(iterations):
    x0=x1=x2=x3=x4=x5=x6=x7=x8=x9=x10=x11=x12=x13=x14=x15=0.1
    a, b = 1.0000001, 0.0000003
    for _ in range(iterations):
        x0=x0*a+b; x1=x1*a+b; x2=x2*a+b; x3=x3*a+b
        x4=x4*a+b; x5=x5*a+b; x6=x6*a+b; x7=x7*a+b
        x8=x8*a+b; x9=x9*a+b; x10=x10*a+b; x11=x11*a+b
        x12=x12*a+b; x13=x13*a+b; x14=x14*a+b; x15=x15*a+b
    return sum((x0,x1,x2,x3,x4,x5,x6,x7,x8,x9,x10,x11,x12,x13,x14,x15))


def gemm_job(args):
    lo, hi, variant, bs = args
    A, B, C = _VIEWS
    n = _N
    # Clearing output is included consistently in all GEMM timings.
    for pos in range(lo*n, hi*n):
        C[pos] = 0.0
    if variant == 'naive':
        for i in range(lo, hi):
            row = i*n
            for j in range(n):
                dot = 0.0
                for k in range(n):
                    dot += A[row+k] * B[k*n+j]
                C[row+j] = dot
    elif variant == 'reordered':
        for i in range(lo, hi):
            row = i*n
            for k in range(n):
                aik, brow = A[row+k], k*n
                for j in range(n):
                    C[row+j] += aik * B[brow+j]
    elif variant == 'blocked':
        for ii in range(lo, hi, bs):
            for jj in range(0, n, bs):
                endj = min(n, jj+bs)
                for kk in range(0, n, bs):
                    for i in range(ii, min(hi, ii+bs)):
                        # Native contiguous views hoisted outside the j loop.
                        crow = C[i*n+jj:i*n+endj]
                        for k in range(kk, min(n, kk+bs)):
                            aik = A[i*n+k]
                            brow = B[k*n+jj:k*n+endj]
                            for j in range(endj-jj):
                                crow[j] += aik * brow[j]
                            brow.release()
                        crow.release()
    else:
        raise ValueError(variant)
    return True


def ranges(n, p):
    return [(t*n//p, (t+1)*n//p) for t in range(p)]


def phase3(p, warmups, runs):
    r = D.setdefault('phase3', {})
    size = 96_000_000
    if 'bandwidth' not in r:
        buffers = [shared_memory.SharedMemory(create=True, size=size) for _ in range(2)]
        try:
            # First touch every page before timing. The source is nonzero.
            for index, handle in enumerate(buffers):
                block = bytes([index+1]) * 1_000_000
                for offset in range(0, size, len(block)):
                    handle.buf[offset:offset+len(block)] = block
            with mp.Pool(p, initializer=attach,
                         initargs=([h.name for h in buffers], 'B')) as pool:
                pool.map(ready, range(p), chunksize=1)
                log('Phase 3A: DRAM copy (96 MB read + 96 MB write)')
                stats = bench(lambda: pool.map(copy_job, ranges(size,p), chunksize=1), warmups,runs)
            stats['GBps'] = 2*size/(stats['median_ms']*1e6)
            if bytes(buffers[0].buf[:64]) != bytes(buffers[1].buf[:64]):
                raise RuntimeError('Memory copy validation failed')
            r['bandwidth'] = stats
            save()
        finally:
            for h in buffers:
                h.close(); h.unlink()
    if 'compute' not in r:
        with mp.Pool(p) as pool:
            pool.map(ready, range(p), chunksize=1)
            log('Phase 3A: 80,000,000 FLOPs per physical worker')
            stats = bench(lambda:pool.map(compute_job,[2_500_000]*p,chunksize=1),warmups,runs)
        stats['GFLOPs'] = p*80_000_000/(stats['median_ms']*1e6)
        r['compute'] = stats
        save()
    n = 384
    buffers = [shared_memory.SharedMemory(create=True, size=n*n*8) for _ in range(3)]
    views = [h.buf.cast('d') for h in buffers]
    try:
        for i in range(n*n):
            views[0][i] = (i % 17 + 1)/17
            views[1][i] = (i % 13 + 1)/13
        rows = r.setdefault('gemm', {})
        with mp.Pool(p, initializer=attach, initargs=([h.name for h in buffers],'d',n)) as pool:
            pool.map(ready, range(p), chunksize=1)
            def run(variant, bs=0):
                pool.map(gemm_job,[(lo,hi,variant,bs) for lo,hi in ranges(n,p)],chunksize=1)
            # Untimed reference; all variants use the same ascending k order.
            run('reordered')
            reference = list(views[2])
            jobs = [('naive',0),('reordered',0)] + [('blocked',b) for b in TILES]
            for variant, bs in jobs:
                key = variant if not bs else f'blocked_{bs}'
                if key in rows:
                    continue
                log(f'Phase 3B: {key} ({warmups} warmups + {runs} timed)')
                stats = bench(lambda:run(variant,bs),warmups,runs)
                error = max(abs(a-b) for a,b in zip(reference,views[2]))
                if error >= 1e-9:
                    raise RuntimeError(f'{key}: maximum error {error} exceeds tolerance')
                stats.update(tile=bs, GFLOPs=2*n**3/(stats['median_ms']*1e6), max_abs_error=error)
                rows[key] = stats
                save()
                write_report()
        best = min((k for k in rows if k.startswith('blocked_')), key=lambda k:rows[k]['median_ms'])
        r['best'] = best
        r['ridge'] = r['compute']['GFLOPs']/r['bandwidth']['GBps']
        r['verify'] = verify('PHASE3',rows['naive']['GFLOPs'],rows[best]['GFLOPs'],mp.cpu_count())
        log(r['verify'])
        save()
    finally:
        for view in views:
            view.release()
        for h in buffers:
            h.close(); h.unlink()
    write_report()


def item_work(i):
    x = (i+1)*1e-5
    for _ in range(1+i*i//3600):
        x = math.sin(x)+math.cos(x)+math.sqrt(abs(x)+1e-12)
    return x


def scheduling_worker(t,p,n,strategy,chunk,counter,elapsed,checksums,barrier):
    barrier.wait()
    start = time.perf_counter_ns()
    checksum = 0.0
    if strategy == 'dynamic':
        while True:
            with counter.get_lock():
                lo = counter.value
                counter.value += chunk
            if lo >= n:
                break
            for i in range(lo,min(n,lo+chunk)):
                checksum += item_work(i)
    else:
        items = range(t*n//p,(t+1)*n//p) if strategy == 'block' else range(t,n,p)
        for i in items:
            checksum += item_work(i)
    elapsed[t] = (time.perf_counter_ns()-start)/1e6
    checksums[t] = checksum


def schedule(p,strategy,chunk,n=12000):
    counter = mp.Value('i',0)
    elapsed, checksums = mp.RawArray('d',p),mp.RawArray('d',p)
    barrier = mp.Barrier(p)
    workers = [mp.Process(target=scheduling_worker,
                         args=(t,p,n,strategy,chunk,counter,elapsed,checksums,barrier)) for t in range(p)]
    start = time.perf_counter_ns()
    for worker in workers:
        worker.start()
    for worker in workers:
        worker.join()
    if any(worker.exitcode != 0 for worker in workers):
        raise RuntimeError('Scheduling worker failed')
    wall = (time.perf_counter_ns()-start)/1e6
    times = list(elapsed)
    return dict(wall_ms=wall,worker_ms=times,imbalance=max(times)/statistics.mean(times),
                checksum=sum(checksums),chunk=chunk)


def phase4(p):
    rows = D.setdefault('phase4',{}).setdefault('rows',{})
    for strategy,chunk in [('block',0),('cyclic',0)]+[('dynamic',c) for c in (1,4,16,64,256,1024)]:
        key = strategy if not chunk else f'dynamic_{chunk}'
        if key in rows:
            continue
        log(f'Phase 4: {key}; N=12,000; one complete measured run')
        result = schedule(p,strategy,chunk)
        if rows:
            reference = next(iter(rows.values()))['checksum']
            if not math.isclose(result['checksum'],reference,rel_tol=1e-10,abs_tol=1e-8):
                raise RuntimeError('Scheduler checksum differs; missing or duplicated work')
        rows[key] = result
        log(f"{key}: wall_ms={result['wall_ms']:.4f}; lambda={result['imbalance']:.6f}; worker_ms={result['worker_ms']}")
        save()
        write_report()
    best = min((k for k in rows if k.startswith('dynamic_')),key=lambda k:rows[k]['wall_ms'])
    r = D['phase4']
    r['best'] = best
    r['verify'] = verify('PHASE4',rows['block']['wall_ms'],rows[best]['wall_ms'],mp.cpu_count())
    log(r['verify'])
    save()
    write_report()

    return '\n'.join(['| '+' | '.join(headers)+' |','|'+'|'.join(['---']*len(headers))+'|']+
                     ['| '+' | '.join(map(str,row))+' |' for row in rows])+'\n'

def self_test():
    n,p = 12,2
    buffers = [shared_memory.SharedMemory(create=True,size=n*n*8) for _ in range(3)]
    views = [h.buf.cast('d') for h in buffers]
    try:
        for i in range(n*n):
            views[0][i]=(i%7)/7; views[1][i]=(i%11)/11
        ref = [sum(views[0][i*n+k]*views[1][k*n+j] for k in range(n)) for i in range(n) for j in range(n)]
        with mp.Pool(p,initializer=attach,initargs=([h.name for h in buffers],'d',n)) as pool:
            for variant,bs in [('naive',0),('reordered',0),('blocked',4),('blocked',8),('blocked',128)]:
                pool.map(gemm_job,[(lo,hi,variant,bs) for lo,hi in ranges(n,p)])
                assert max(abs(a-b) for a,b in zip(ref,views[2])) < 1e-9
        reference = None
        for kind,chunk in [('block',0),('cyclic',0),('dynamic',1),('dynamic',16)]:
            result = schedule(p,kind,chunk,n=120)
            if reference is not None:
                assert math.isclose(reference,result['checksum'],rel_tol=1e-10)
            reference = result['checksum']
        print('SELF TEST PASSED: GEMM variants, partial tiles, schedulers and checksums.')
    finally:
        for v in views: v.release()
        for h in buffers: h.close(); h.unlink()
